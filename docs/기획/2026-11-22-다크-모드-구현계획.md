```ts
/**
 * 다크 모드 (docs/기획/2026-11-22-다크-모드-설계.md §1).
 *
 * 고른 값은 이 기기에만(perky.theme) — 서버로 보내지 않고 로그아웃해도 남긴다(온보딩 본 기록 같은 기기 설정).
 * '기기 설정'(system)은 저장값을 지운 상태다. 적용은 <html data-theme> · colorScheme · <meta name="theme-color">.
 * 그리기 전에는 <head> 의 THEME_SCRIPT, 그 뒤 바뀜은 ThemeWatcher 가 applyTheme 으로 다시 건다.
 */

export const THEME_KEY = "perky.theme";
export type ThemeChoice = "system" | "light" | "dark";
export type ResolvedTheme = "light" | "dark";
/** 상단 상태 표시줄 · 브라우저 막대 색 = 페이지 바탕(canvas) */
export const THEME_COLORS: Record<ResolvedTheme, string> = { light: "#f4f8ff", dark: "#0c1620" };
/** 이 탭에서 고른 값이 바뀌었다 — ThemeWatcher · 내 정보 화면이 듣는다 */
export const THEME_CHANGE_EVENT = "perky:theme";
const DARK_QUERY = "(prefers-color-scheme: dark)";

function isThemeChoice(value: unknown): value is ThemeChoice {
  return value === "system" || value === "light" || value === "dark";
}

export function resolveTheme(choice: ThemeChoice, prefersDark: boolean): ResolvedTheme {
  return choice === "dark" || (choice === "system" && prefersDark) ? "dark" : "light";
}

export function loadThemeChoice(): ThemeChoice {
  try {
    const value = window.localStorage.getItem(THEME_KEY);
    return isThemeChoice(value) ? value : "system";
  } catch {
    return "system";
  }
}

function prefersDark(): boolean {
  try {
    return window.matchMedia?.(DARK_QUERY).matches === true;
  } catch {
    return false;
  }
}

export function applyTheme(choice: ThemeChoice = loadThemeChoice()): ResolvedTheme {
  const theme = resolveTheme(choice, prefersDark());
  const root = document.documentElement;
  root.dataset.theme = theme;
  root.style.colorScheme = theme;
  document.querySelector('meta[name="theme-color"]')?.setAttribute("content", THEME_COLORS[theme]);
  return theme;
}

export function saveThemeChoice(choice: ThemeChoice): void {
  try {
    if (choice === "system") window.localStorage.removeItem(THEME_KEY);
    else window.localStorage.setItem(THEME_KEY, choice);
  } catch {
    // 저장이 막혀 있어도 이번 화면에는 적용한다
  }
  applyTheme(choice);
  window.dispatchEvent(new Event(THEME_CHANGE_EVENT));
}

/** 기기의 다크 설정이 바뀔 때. matchMedia 가 없으면 아무것도 안 한다. */
export function onSystemThemeChange(listener: () => void): () => void {
  const query = typeof window.matchMedia === "function" ? window.matchMedia(DARK_QUERY) : null;
  if (!query?.addEventListener) return () => {};
  query.addEventListener("change", listener);
  return () => query.removeEventListener("change", listener);
}

/**
 * 번들보다 먼저 도는 인라인 스크립트 — import 없이 값을 글자로 박는다(lib/splash.ts 와 같은 방식, 시험이 실제로 돌려 본다).
 * 실패하면 아무것도 안 한다 — 그러면 CSS 기본(밝게)이다.
 */
export const THEME_SCRIPT = `try{var c=localStorage.getItem(${JSON.stringify(THEME_KEY)});var d=c==="dark"||(c!=="light"&&matchMedia(${JSON.stringify(
  DARK_QUERY
)}).matches);var r=document.documentElement;r.dataset.theme=d?"dark":"light";r.style.colorScheme=d?"dark":"light";var m=document.querySelector('meta[name="theme-color"]');if(m)m.setAttribute("content",d?${JSON.stringify(
  THEME_COLORS.dark
)}:${JSON.stringify(THEME_COLORS.light)})}catch(e){}`;
```

```tsx
"use client";

import { useEffect } from "react";
import { THEME_CHANGE_EVENT, THEME_KEY, applyTheme, onSystemThemeChange } from "@/lib/theme";

/**
 * 그린 뒤의 테마 바뀜을 따라간다(그리기 전은 layout <head> 의 THEME_SCRIPT).
 * 기기 다크 설정 바뀜 · 다른 탭에서 고름(storage) · 이 탭에서 고름(THEME_CHANGE_EVENT). 화면에는 아무것도 그리지 않는다.
 */
export function ThemeWatcher() {
  useEffect(() => {
    const reapply = () => applyTheme();
    const onStorage = (event: StorageEvent) => {
      if (event.key === THEME_KEY || event.key === null) reapply();
    };
    const offSystem = onSystemThemeChange(reapply);
    window.addEventListener("storage", onStorage);
    window.addEventListener(THEME_CHANGE_EVENT, reapply);
    return () => {
      offSystem();
      window.removeEventListener("storage", onStorage);
      window.removeEventListener(THEME_CHANGE_EVENT, reapply);
    };
  }, []);
  return null;
}
```

```ts
import { readFileSync } from "node:fs";
import { join } from "node:path";
import { describe, expect, it } from "vitest";

/** globals.css 를 그대로 읽어 밝게(@theme) · 어둡게([data-theme="dark"] 덮개) 토큰을 뽑고 대비를 잰다 — 값만 바꾸고 다시 안 재는 일이 없게. */
const css = readFileSync(join(__dirname, "../../app/globals.css"), "utf8");

function tokens(open: RegExp): Record<string, string> {
  const m = open.exec(css);
  if (!m) throw new Error(`블록 없음: ${open}`);
  let i = m.index + m[0].length;
  let depth = 1;
  const start = i;
  while (depth > 0 && i < css.length) {
    if (css[i] === "{") depth++;
    else if (css[i] === "}") depth--;
    i++;
  }
  const out: Record<string, string> = {};
  for (const [, name, hex] of css.slice(start, i - 1).matchAll(/--color-([a-z0-9-]+):\s*(#[0-9a-fA-F]{6})\b/g)) out[name] = hex.toLowerCase();
  return out;
}

const light = tokens(/@theme\s*\{/);
const dark = { ...light, ...tokens(/^\[data-theme="dark"\]\s*\{/m) };

function luminance(hex: string): number {
  const [r, g, b] = [1, 3, 5].map((i) => parseInt(hex.slice(i, i + 2), 16) / 255).map((c) => (c <= 0.03928 ? c / 12.92 : ((c + 0.055) / 1.055) ** 2.4));
  return 0.2126 * r + 0.7152 * g + 0.0722 * b;
}
function contrast(a: string, b: string): number {
  const [hi, lo] = [luminance(a), luminance(b)].sort((x, y) => y - x);
  return (hi + 0.05) / (lo + 0.05);
}
const color = (set: Record<string, string>, name: string) => (name === "white" ? "#ffffff" : set[name]);

/** [글자/아이콘, 바탕, 최소 대비] — 토큰 주석의 쓰임대로 */

  ["ink-500", "surface", 4.5], ["ink-500", "canvas", 4.5],

  ["ok-700", "surface", 4.5], ["ok-700", "ok-50", 4.5], ["warn-800", "warn-50", 4.5],
  ["danger-700", "surface", 4.5], ["danger-700", "danger-50", 4.5],
  ["white", "brand-600", 4.5], ["white", "brand-press", 4.5], ["white", "danger-600", 4.5], ["white", "danger-press", 4.5],
  ["brand-600", "surface", 3],
];

describe.each([
```

```tsx
"use client";

import { useEffect, useState } from "react";
import { THEME_CHANGE_EVENT, loadThemeChoice, saveThemeChoice, type ThemeChoice } from "@/lib/theme";

const OPTIONS: { value: ThemeChoice; label: string }[] = [
  { value: "system", label: "기기 설정" },
  { value: "light", label: "밝게" },
  { value: "dark", label: "어둡게" },
];

/**
 * 화면 모드 (docs/기획/2026-11-22-다크-모드-설계.md §3). 로그인과 상관없이 — 이 기기에만 저장한다.
 * 세 칸 중 하나를 누르는 버튼 묶음(aria-pressed). 고른 칸은 파랑 채움 + 굵게 — 색만으로 말하지 않는다.
 */
export function ThemeSetting() {
  const [choice, setChoice] = useState<ThemeChoice>("system");

  useEffect(() => {
    const read = () => setChoice(loadThemeChoice());
    read();
    window.addEventListener(THEME_CHANGE_EVENT, read);
    return () => window.removeEventListener(THEME_CHANGE_EVENT, read);
  }, []);

  return (
    <section aria-labelledby="theme-heading" className="flex flex-col gap-2.5">
      <h2 id="theme-heading" className="text-base font-bold text-ink-900">
        화면 모드
      </h2>
      <div role="group" aria-labelledby="theme-heading" className="grid grid-cols-3 gap-1 rounded-control bg-ink-100 p-1">
        {OPTIONS.map((option) => {
          const on = option.value === choice;
          return (
            <button
              key={option.value}
              type="button"
              aria-pressed={on}
              onClick={() => {
                saveThemeChoice(option.value);
                setChoice(option.value);
              }}
              className={`focus-ring min-h-11 rounded-[12px] text-sm transition-colors ${
                on ? "bg-brand-600 font-bold text-white" : "font-semibold text-ink-600 hover:bg-surface hover:text-ink-900"
              }`}
            >
              {option.label}
            </button>
          );
        })}
      </div>
      <p className="text-xs leading-relaxed text-ink-500">기기 설정을 고르면 휴대폰이 어두운 모드일 때 같이 어두워져요. 이 기기에만 저장돼요.</p>
    </section>
  );
}
```
