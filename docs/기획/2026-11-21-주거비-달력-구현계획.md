### Task 1: 계산 규칙 `lib/ledger.ts`

**Files:**
- Create: `apps/web/lib/ledger.ts`
- Test: `apps/web/lib/__tests__/ledger.test.ts`

**Interfaces:**
- Consumes: `monthlyBreakdown(policy, listing): { monthly: number; months: number } | null` (`lib/benefit.ts`), `formatKoreanMoney` (`lib/money.ts`), 타입 `CalculationSummary · ContractType · ListingInput · PolicyMeta` (`lib/types.ts`)
- Produces (뒤 태스크가 쓰는 이름 그대로):
  - `type MonthKey = string` ("2026-11")
  - `interface LedgerMonthRecord { done: Record<string, number>; amounts: Record<string, number> }`
  - `interface LedgerRecord { v: 1; months: Record<MonthKey, LedgerMonthRecord>; oneTime: Record<string, MonthKey> }`
  - `emptyLedger(): LedgerRecord` · `parseLedger(raw: unknown): LedgerRecord`
  - `monthKeyOf(iso: string): MonthKey` · `addMonths(key, n): MonthKey` · `monthsBetween(from, to): number` · `monthLabel(key): string`("2026년 11월") · `monthOnlyLabel(key): string`("11월") · `receivedMonthLabel(month, todayMonth): string` · `yearTotalLabel(month, todayMonth): string`
  - `RENT_KEY = "rent"` · `FEE_KEY = "fee"` · `policyLineKey(id): string`
  - `interface LedgerPlan { start; end; months; contractType; rent; fee; incomes: { policy: PolicyMeta; monthly: number; months: number }[] }` · `ledgerPlan(listing, summary): LedgerPlan`
  - `interface LedgerLine { key; side: "out" | "in"; label; note: string | null; planned; amount; edited; done }`
  - `interface LedgerMonthView { month; inContract; contractMonth; lines; actual; expected; pending; yearTotal }` · `ledgerMonth(plan, record, month): LedgerMonthView`
  - `toggleLine(record, month, key, amount): LedgerRecord` · `setLineAmount(record, month, line: Pick<LedgerLine, "key" | "planned">, won: number | null): LedgerRecord` · `toggleOneTime(record, policyId, month): LedgerRecord`
  - `interface OneTimeSupport { policy: PolicyMeta; amountText: string | null }` · `oneTimeSupports(summary): OneTimeSupport[]`
  - `interface LedgerProgress { outDone; outTotal; inDone; inTotal; complete }` · `ledgerProgress(view): LedgerProgress | null`

import type { CalculationSummary, PolicyMeta, PolicyResult, PolicyStatus } from "../types";
import { makeListing } from "./fixtures";

const policies = policiesJson as PolicyMeta[];
function real(id: string): PolicyMeta {
  const p = policies.find((x) => x.id === id);
  if (!p) throw new Error(`정책 없음: ${id}`);
  return p;

// 값이 DB 스냅숏에 흔들리지 않게 계산에 쓰는 칸은 여기서 고정한다

const 익산월세: PolicyMeta = { ...real("iksan-youth-rent-support"), benefitType: "rent_capped_monthly", monthlyCap: 200_000, maxMonths: 3 };
const 전입장려금: PolicyMeta = { ...real("iksan-move-in-household-incentive"), benefitType: "lump_sum", lumpSumBasis: "fixedGrant", lumpSumCap: 100_000 };

function result(policy: PolicyMeta, status: PolicyStatus = "예상적용"): PolicyResult {
  return { policy, status, checks: [], passedLabels: [], failedLabels: [], unknownLabels: [], estimatedAmount: 0 };
}
function summary(results: PolicyResult[], included: string[]): CalculationSummary {
  return { nominalTotalCost: 0, maxSupportAmount: 0, finalEstimatedHousingCost: 0, bestCombination: { includedPolicyIds: included, totalAmount: 0 }, results };
}

// 시안 예시: 월세 40만 · 관리비 5만 · 2026년 9월 시작 12개월 · 청년 주거급여 분리지급 월 21만 2천원

const 월세계획 = ledgerPlan(월세, summary([result(주거급여)], [주거급여.id]));
const 급여 = policyLineKey(주거급여.id);

/** 그 달 칸을 지금 금액으로 누른다 */
function press(record: LedgerRecord, month: string, ...keys: string[]): LedgerRecord {
  let next = record;
  for (const key of keys) {
    const line = ledgerMonth(월세계획, next, month).lines.find((l) => l.key === key);
    if (!line) throw new Error(`${month} 에 ${key} 칸 없음`);
    next = toggleLine(next, month, key, line.amount);
  }
  return next;
}

/**
 * 주거비 달력 (docs/기획/2026-11-21-주거비-달력-설계.md §4 · §5).
 *
 * 계산할 때 넣은 계약 조건과 결과 화면의 지원 조합으로 달마다 낼 돈 · 받을 지원을 미리 적는다(계획, LedgerPlan).
 * 사람이 한 일 — 누른 칸(누를 때 금액) · 고친 금액 · 한 번 받는 지원의 받은 달 — 만 기록(LedgerRecord)으로 남기고,
 * 화면 값은 열 때마다 계획 + 기록으로 다시 센다. 그래서 다시 계산하면 안 누른 칸만 저절로 새 값이 된다(결정 ⑤).
 * 이 파일은 저장소를 모른다(순수 함수). 읽고 쓰기는 lib/storage.ts 의 loadLedger · saveLedger.
 */

/** "2026-11" */
export type MonthKey = string;

export interface LedgerMonthRecord {
  /** 누른 칸 → 누를 때의 금액(원). 다시 계산해도 누른 칸 금액은 그대로 둔다. */
  done: Record<string, number>;
  /** 고친 칸 → 고친 금액(원). 이 달에만. */
  amounts: Record<string, number>;
}

export interface LedgerRecord {
  v: 1;
  months: Record<MonthKey, LedgerMonthRecord>;
  /** 한 번 받는 지원: 정책 id → 받은 달 */
  oneTime: Record<string, MonthKey>;
}

export function emptyLedger(): LedgerRecord {
  return { v: 1, months: {}, oneTime: {} };
}

const MONTH_RE = /^\d{4}-(0[1-9]|1[0-2])$/;

function positiveAmounts(raw: unknown): Record<string, number> {
  const out: Record<string, number> = {};
  if (!raw || typeof raw !== "object" || Array.isArray(raw)) return out;
  for (const [key, value] of Object.entries(raw as Record<string, unknown>)) {
    if (typeof value === "number" && Number.isFinite(value) && value > 0) out[key] = Math.round(value);
  }
  return out;
}

/** 저장소에서 읽은 값을 기록으로. 모양이 틀린 부분은 버린다(다른 저장값처럼 조용히). */
export function parseLedger(raw: unknown): LedgerRecord {
  const out = emptyLedger();
  if (!raw || typeof raw !== "object") return out;
  const r = raw as { v?: unknown; months?: unknown; oneTime?: unknown };
  if (r.v !== 1) return out;
  if (r.months && typeof r.months === "object") {
    for (const [month, value] of Object.entries(r.months as Record<string, unknown>)) {
      if (!MONTH_RE.test(month) || !value || typeof value !== "object") continue;
      const m = value as { done?: unknown; amounts?: unknown };
      out.months[month] = { done: positiveAmounts(m.done), amounts: positiveAmounts(m.amounts) };
    }
  }
  if (r.oneTime && typeof r.oneTime === "object") {
    for (const [id, month] of Object.entries(r.oneTime as Record<string, unknown>)) {
      if (typeof month === "string" && MONTH_RE.test(month)) out.oneTime[id] = month;
    }
  }
  return out;
}

// ── 달 ──

export function monthKeyOf(iso: string): MonthKey {
  return iso.slice(0, 7);
}

function toIndex(key: MonthKey): number {
  return Number(key.slice(0, 4)) * 12 + Number(key.slice(5, 7)) - 1;
}

function fromIndex(index: number): MonthKey {
  const year = Math.floor(index / 12);
  return `${year}-${String(index - year * 12 + 1).padStart(2, "0")}`;
}

export function addMonths(key: MonthKey, n: number): MonthKey {
  return fromIndex(toIndex(key) + n);
}

/** to − from (개월). */
export function monthsBetween(from: MonthKey, to: MonthKey): number {
  return toIndex(to) - toIndex(from);
}

/** "2026년 11월" */
export function monthLabel(key: MonthKey): string {
  return `${Number(key.slice(0, 4))}년 ${Number(key.slice(5, 7))}월`;
}

/** "11월" */
export function monthOnlyLabel(key: MonthKey): string {
  return `${Number(key.slice(5, 7))}월`;
}

/** 한 번 받는 지원의 "10월에 받음" — 해가 다르면 "2025년 12월". */
export function receivedMonthLabel(month: MonthKey, todayMonth: MonthKey): string {
  return month.slice(0, 4) === todayMonth.slice(0, 4) ? monthOnlyLabel(month) : monthLabel(month);
}

/** 보고 있는 달이 올해가 아니면 '올해'라고 하지 않는다 — 2027년 1월을 보며 '올해'라면 2026년으로 읽힌다. */
export function yearTotalLabel(month: MonthKey, todayMonth: MonthKey): string {
  return month.slice(0, 4) === todayMonth.slice(0, 4) ? "올해 누적" : `${Number(month.slice(0, 4))}년 누적`;
}

export const RENT_KEY = "rent";
export const FEE_KEY = "fee";

export function policyLineKey(id: string): string {
  return `policy:${id}`;

export interface LedgerIncome {
  policy: PolicyMeta;
  monthly: number;
  /** 계약 1개월째부터 이 개월째까지 들어온다 */
  months: number;
}

export interface LedgerPlan {
  start: MonthKey;
  end: MonthKey;
  months: number;
  contractType: ContractType;
  /** 월세 또는 연세 */
  rent: number;
  fee: number;
  incomes: LedgerIncome[];
}

/** 들어오는 돈 = 결과 합계(최선 조합)에 든 월 지원. 일시금은 따로(oneTimeSupports). */
export function ledgerPlan(listing: ListingInput, summary: CalculationSummary): LedgerPlan {
  const start = monthKeyOf(listing.contractStartDate);
  const months = Math.max(1, listing.months);
  const included = new Set(summary.bestCombination.includedPolicyIds);
  const incomes: LedgerIncome[] = [];
  for (const result of summary.results) {
    if (!included.has(result.policy.id)) continue;
    const breakdown = monthlyBreakdown(result.policy, listing);
    if (breakdown && breakdown.monthly > 0) incomes.push({ policy: result.policy, monthly: breakdown.monthly, months: breakdown.months });
  }
  return {
    start,
    end: addMonths(start, months - 1),
    months,
    contractType: listing.contractType,
    rent: listing.rentOrYearlyAmount,
    fee: listing.managementFee,
    incomes,
  };
}

export interface LedgerLine {
  key: string;
  side: "out" | "in";
  label: string;
  /** 연세 줄의 보조문 */
  note: string | null;
  /** 지금 계산값 */
  planned: number;
  /** 화면 금액 = 고친 값 → 누를 때 금액 → 계산값 */
  amount: number;
  edited: boolean;
  done: boolean;
}

export interface LedgerMonthView {
  month: MonthKey;
  inContract: boolean;
  /** 계약 N개월째 (기간 밖이면 1 미만이거나 months 초과) */
  contractMonth: number;
  lines: LedgerLine[];
  /** 누른 나가는 돈 − 누른 들어오는 돈. 음수 그대로. */
  actual: number;
  /** 모든 칸을 계산값으로 다 눌렀을 때. 고친 값은 넣지 않는다. 음수 그대로. */
  expected: number;
  /** 안 누른 칸이 있다 */
  pending: boolean;
  /** 그 해 1월 ~ 이 달의 actual 합. 음수 그대로. */
  yearTotal: number;
}

function linesFor(plan: LedgerPlan, record: LedgerRecord, month: MonthKey): LedgerLine[] {
  const n = monthsBetween(plan.start, month) + 1;
  if (n < 1 || n > plan.months) return [];
  const pressed = record.months[month]?.done ?? {};
  const amounts = record.months[month]?.amounts ?? {};
  const line = (key: string, side: LedgerLine["side"], label: string, planned: number, note: string | null = null): LedgerLine => {
    const edited = amounts[key] !== undefined;
    const done = pressed[key] !== undefined;
    return { key, side, label, note, planned, amount: edited ? amounts[key] : done ? pressed[key] : planned, edited, done };
  };
  const lines: LedgerLine[] = [];
  if (plan.contractType === "연세") {
    if (n === 1 && plan.rent > 0) lines.push(line(RENT_KEY, "out", "연세", plan.rent, "연세는 계약 시작 달에 한 번만 나와요"));
  } else if (plan.rent > 0) {
    lines.push(line(RENT_KEY, "out", "월세", plan.rent));
  }
  if (plan.fee > 0) lines.push(line(FEE_KEY, "out", "관리비", plan.fee));
  for (const income of plan.incomes) {
    if (n <= income.months) lines.push(line(policyLineKey(income.policy.id), "in", income.policy.name, income.monthly));
  }
  return lines;
}

const sign = (l: LedgerLine) => (l.side === "out" ? 1 : -1);

function actualOf(lines: LedgerLine[]): number {
  return lines.reduce((sum, l) => (l.done ? sum + sign(l) * l.amount : sum), 0);
}

export function ledgerMonth(plan: LedgerPlan, record: LedgerRecord, month: MonthKey): LedgerMonthView {
  const contractMonth = monthsBetween(plan.start, month) + 1;
  const lines = linesFor(plan, record, month);
  let yearTotal = 0;
  for (let m = `${month.slice(0, 4)}-01`; monthsBetween(m, month) >= 0; m = addMonths(m, 1)) {
    yearTotal += actualOf(linesFor(plan, record, m));
  }
  return {
    month,
    inContract: contractMonth >= 1 && contractMonth <= plan.months,
    contractMonth,
    lines,
    actual: actualOf(lines),
    expected: lines.reduce((sum, l) => sum + sign(l) * l.planned, 0),
    pending: lines.some((l) => !l.done),
    yearTotal,
  };
}

// ── 기록 바꾸기 (원래 기록은 그대로 두고 새 기록을 돌려준다) ──

function withMonth(record: LedgerRecord, month: MonthKey, edit: (m: LedgerMonthRecord) => LedgerMonthRecord): LedgerRecord {
  const current = record.months[month] ?? { done: {}, amounts: {} };
  return { ...record, months: { ...record.months, [month]: edit(current) } };
}

/** 누르면 그때 금액을 남기고, 다시 누르면 지운다. */
export function toggleLine(record: LedgerRecord, month: MonthKey, key: string, amount: number): LedgerRecord {
  return withMonth(record, month, (m) => {
    const done = { ...m.done };
    if (done[key] !== undefined) delete done[key];
    else done[key] = Math.round(amount);
    return { ...m, done };
  });
}

/**
 * 이 달 이 칸의 금액을 고친다. 비우거나(null · 0 이하) 계산값과 같으면 고친 값을 지운다(= [계산값으로]).
 * 이미 누른 칸이면 누른 금액도 새 금액으로 따라간다 — 안 그러면 [계산값으로] 뒤에도 고친 금액이 남는다.
 */
export function setLineAmount(
  record: LedgerRecord,
  month: MonthKey,
  line: Pick<LedgerLine, "key" | "planned">,
  won: number | null
): LedgerRecord {
  return withMonth(record, month, (m) => {
    const value = won !== null && won > 0 && Math.round(won) !== line.planned ? Math.round(won) : null;
    const amounts = { ...m.amounts };
    if (value === null) delete amounts[line.key];
    else amounts[line.key] = value;
    const done = { ...m.done };
    if (done[line.key] !== undefined) done[line.key] = value ?? line.planned;
    return { done, amounts };
  });
}

/** 한 번 받는 지원 — 누르면 받은 달을 남기고, 다시 누르면 지운다. */
export function toggleOneTime(record: LedgerRecord, policyId: string, month: MonthKey): LedgerRecord {
  const oneTime = { ...record.oneTime };
  if (oneTime[policyId] !== undefined) delete oneTime[policyId];
  else oneTime[policyId] = month;
  return { ...record, oneTime };
}

export interface LedgerProgress {
  outDone: number;
  outTotal: number;
  inDone: number;
  inTotal: number;
  complete: boolean;
}

/** 그 달 누른 개수. 계약 기간 밖이거나 칸이 없으면 null(홈 한 줄을 숨긴다). */
export function ledgerProgress(view: LedgerMonthView): LedgerProgress | null {

}

/**
 * 하단 섬 탭 (디자인 캔버스 '하단 섬 · 상단바 설계').
 *
 * 화면 아래에 떠 있는 둥근 막대다. 홈 · 관심 · 지원금 · 달력 · 내 정보처럼 '탭의 첫 화면'과 그 하위 화면(결과 → 홈에 불)에 둔다.

 * 그 화면들은 아래에 고정 버튼(StickyBottomAction)이 이미 있다.
 *
 * 로그인한 사람에게만 보인다. 홈 · 관심 · 달력은 로그인한 사람만 들어갈 수 있어서, 로그인 전에 탭을 보이면

 *

    if (!isLoginAvailable()) return;
    const update = () => setSignedIn(hasStoredLogin());
    update();
    return onAuthChange(update);
  }, []);

  if (!signedIn) return null;
  const failed = sync.status === "error";

  return (

      <div aria-hidden="true" data-testid="tab-bar-spacer" className="h-[calc(6.5rem+env(safe-area-inset-bottom))] shrink-0" />
      <nav
        aria-label="주요 메뉴"
        className="pointer-events-none fixed inset-x-0 bottom-0 z-30 pb-[max(1.25rem,env(safe-area-inset-bottom))]"
      >
        <div className="pointer-events-auto mx-auto flex w-[calc(100%-2.5rem)] max-w-[380px] gap-0.5 rounded-full bg-surface/80 p-1 shadow-island backdrop-blur-xl backdrop-saturate-150">
          {TABS.map((tab) => {
            const on = tab.key === active;
            const alert = tab.key === "my" && failed;
            return (
              <Link
                key={tab.key}
                href={tab.href}
                aria-current={on ? "page" : undefined}
                className={`focus-ring relative flex min-h-[54px] min-w-0 flex-1 flex-col items-center justify-center gap-[3px] rounded-full transition-colors ${
                  on ? "bg-brand-600/12 text-brand-600" : "text-ink-600 hover:bg-ink-100/70 active:bg-ink-200/70"
                }`}
              >
                <span className="relative">
                  <TabIcon name={tab.icon} filled={on} />
                  {alert && <span aria-hidden="true" className="absolute -right-1 -top-0.5 size-2 rounded-full bg-warn-800 ring-2 ring-surface" />}
                </span>
                <span className={`text-[12px] leading-tight ${on ? "font-bold text-brand-700" : "font-semibold"}`}>
                  {tab.label}
                  {alert && <span className="sr-only"> (저장 안 됨)</span>}
                </span>
              </Link>
            );
          })}
        </div>
      </nav>
    </>
  );
}

const policies = policiesJson as PolicyMeta[];

/**
 * 주거비 달력 — 계산한 값이 미리 적힌 체크표 (docs/기획/2026-11-21-주거비-달력-설계.md, 하단 섬 '달력' 탭).
 * 로그인한 사람만(관심 화면과 같은 방식). 기록은 이 기기에만(lib/storage.ts loadLedger · saveLedger).
 * 보고 있는 달은 화면 상태로만 둔다 — 주소에 남기면 뒤로 가기가 달을 하나씩 되짚는다.
 * 계산값이 바뀌면(onLocalChange — 다른 탭에서 다시 계산 · 서버 값 받기) 계획을 다시 만든다. 기록은 그대로.
 */
export default function LedgerPage() {
  const [ready, setReady] = useState(false);
  const [calc, setCalc] = useState<Calculated | null>(null);
  const [record, setRecord] = useState<LedgerRecord | null>(null);
  const [todayMonth, setTodayMonth] = useState<MonthKey>("");
  const [month, setMonth] = useState<MonthKey>("");

  useEffect(() => {
    const leave = () => replacePage("/onboarding");
    if (!isLoginAvailable() || !hasStoredLogin()) {
      leave();
      return;
    }
    const read = () => {
      const today = seoulTodayISO();
      const listing = loadListing();
      const profile = loadProfile();
      let next: Calculated | null = null;
      if (listing && profile) {
        try {
          const summary = buildCalculationSummary(policiesForRegion(policies, listing.region), profile, listing, today);
          next = { plan: ledgerPlan(listing, summary), oneTime: oneTimeSupports(summary) };
        } catch {
          next = null;
        }
      }
      setCalc(next);
      setRecord(loadLedger());
      setTodayMonth(monthKeyOf(today));
      setMonth((current) => current || monthKeyOf(today));
      setReady(true);
    };
    read();
    const checkLogin = () => {
      if (!hasStoredLogin()) leave();
    };
    // 다른 탭의 로그아웃은 lib/auth.ts 의 알림이 오지 않는다 — 브라우저의 storage 이벤트로 듣는다 (관심 화면과 같다).
    const onStorage = (event: StorageEvent) => {
      if (event.key === AUTH_KEY || event.key === null) checkLogin();
    };
    window.addEventListener("storage", onStorage);
    const offAuth = onAuthChange(checkLogin);
    const offLocal = onLocalChange(read);
    return () => {
      window.removeEventListener("storage", onStorage);
      offAuth();
      offLocal();
    };
  }, []);

  const update = useCallback((next: LedgerRecord) => {
    setRecord(next);
    saveLedger(next);
  }, []);

  if (!ready || record === null) {
    return (
      <AppShell>
        <TopBar />
        <p className="mt-10 text-center text-sm text-ink-500">불러오는 중…</p>
      </AppShell>
    );
  }

  return (
    <AppShell>
      <TopBar />
      <main className="flex flex-col pb-10 pt-2.5">
        <h1 className="text-2xl font-extrabold leading-snug text-ink-900">주거비 달력</h1>
        {calc === null ? (
          <BeforeCalculation />
        ) : (
          <>
            <p className="mt-1.5 text-sm leading-normal text-ink-600">계산한 값을 미리 채워 뒀어요. 내고 받을 때 눌러 주세요.</p>
            <MonthSection plan={calc.plan} record={record} month={month} todayMonth={todayMonth} onMonth={setMonth} onChange={update} />
            <OneTimeSection supports={calc.oneTime} record={record} month={month} todayMonth={todayMonth} onChange={update} />
            <p className="mt-5 flex items-start gap-1.5 text-[13px] leading-normal text-ink-500">
              <Info size={ICON_SM} strokeWidth={2} aria-hidden="true" className="mt-0.5 shrink-0" />
              <span>기록은 이 기기에만 저장돼요. 로그아웃하면 지워져요.</span>
            </p>
          </>
        )}
      </main>
      <BottomTabBar active="ledger" />
    </AppShell>
  );
}
