# 주택청약 구현 계획 — 청약 가이드 · 익산 청약 공고 `/housing` · 알림함 · 휴대폰 알림

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 익산 청년이 청약을 이해하고(가이드), 지금 넣을 수 있는 익산 청약 공고를 한 화면에서 보고(`/housing`), 새 공고를 알림함과 휴대폰으로 받게 한다.

**Architecture:** 빌드 전에 `scripts/fetch-notices.mjs` 가 공공데이터(마이홈포털 · 청약홈)에서 익산 공고를 받아 `data/notices.json` 으로 굽는다(실패하면 커밋된 스냅숏). 앱은 그 파일을 읽어 화면 · 알림함을 그리고, `/notices.json` 정적 경로로 내보낸 같은 파일을 알림 서버(AWS, 매시)가 읽어 새 공고를 푸시한다. 하루 한 번 새로 받도록 Supabase 예약 작업이 기존 재빌드 지문을 매일 지운다.

**Tech Stack:** Next.js 15 정적 export · TypeScript · Tailwind 4 · Vitest(jsdom) · Node 22 스크립트(.mjs) · AWS Lambda(apps/api, web-push) · Supabase(pg_cron · PGlite 시험)

**Spec:** `docs/기획/2026-11-22-청약-공고-설계.md`

## Global Constraints

- **F0-5 추정 금지** — 가이드에 확인 못 한 숫자(나이 · 소득 · 금리 · 기간)를 쓰지 않는다. 공고는 **자격을 판정하지 않는다**(배지 · '신청 가능' 같은 말 금지). 공고문 링크와 기간만
- 출처 · 바로가기 주소는 https 공공 주소만(`go.kr` · `gov.kr` · `or.kr` · `korea.kr`) — 예외는 한국부동산원 **청약홈 `www.applyhome.co.kr`** 하나(Task 2 가 시험 규칙에 이름으로 더한다)
- 키는 **`BOJOGEUM_KEY`** 그대로(공공데이터포털 일반 인증키, 11/22 두 API 활용신청 승인 · Vercel Production · Preview 에 이미 있음 — `fetch-civil-places.mjs` 가 쓴다). 새 환경변수를 만들지 않는다. 키 · 키가 든 주소를 로그에 남기지 않는다
- 마이홈 익산 = `brtcCode=52` · `signguCode=140` (옛 `45` 는 NODATA). 청약홈 = 공급지역 `전북` + 주소에 `익산`

```js
// 익산 청약 공고 모으기의 순수 함수 (I/O 는 ../fetch-notices.mjs). 설계: docs/기획/2026-11-22-청약-공고-설계.md §0 · §1
// 11/22 시험: 마이홈은 익산 코드로 바로, 청약홈은 지역 코드가 없어 공급지역 '전북' → 주소 '익산'.

export const MYHOME = "https://apis.data.go.kr/1613000/HWSPR02";
export const APPLYHOME = "https://api.odcloud.kr/api/ApplyhomeInfoDetailSvc/v1";
export const MYHOME_BRTC = "52"; // 전북특별자치도 (옛 45 는 NODATA)
export const MYHOME_SIGNGU = "140"; // 익산시
const ROWS = 100;
const MAX_PAGES = 40; // 청약홈이 지역 조건을 무시해도 전국(약 3천 건)을 다 받을 수 있게

const text = (v) => (typeof v === "string" && v.trim() ? v.trim() : null);
const link = (v) => (typeof v === "string" && /^https?:\/\//.test(v.trim()) ? v.trim() : null);

/** "20261123" · "2026-11-23" → "2026-11-23", 그 밖은 null */
export function ymd(raw) {
  const m = /^(\d{4})-?(\d{2})-?(\d{2})$/.exec(String(raw ?? "").trim());
  return m ? `${m[1]}-${m[2]}-${m[3]}` : null;
}

/** 마이홈 응답의 줄 — 한 건이면 객체로, NODATA 면 body 없이 온다 */
export function myhomeRows(json) {
  const body = json?.response?.body ?? json?.body;
  const items = body?.item ?? body?.items?.item ?? body?.items ?? [];
  return Array.isArray(items) ? items : [items];
}

/** kind: "공공임대"(rsdtRcritNtcList) · "공공분양"(ltRsdtRcritNtcList) */
export function fromMyhome(row, kind) {
  const title = text(row?.pblancNm);
  const url = link(row?.url) ?? link(row?.pcUrl) ?? link(row?.mobileUrl);
  if (!row?.pblancId || !title || !url) return null;
  return {
    id: `myhome:${row.pblancId}`,
    source: "마이홈포털",
    kind,
    typeLabel: kind === "공공임대" ? (text(row.suplyTyNm) ?? "공공임대") : "공공분양",
    title,
    supplier: text(row.suplyInsttNm),
    announcedAt: ymd(row.rcritPblancDe),
    applyStart: ymd(row.beginDe),
    applyEnd: ymd(row.endDe),
    url,
    address: text(row.fullAdres),
    replaces: text(row.beforePblancId) ? `myhome:${row.beforePblancId}` : null,
  };
}

/**
 * kind: "아파트 분양"(getAPTLttotPblancDetail) · "무순위 · 잔여"(getRemndrLttotPblancDetail).
 * 접수 = 특별공급 · 일반 접수 시작 중 이른 날 ~ 접수 마감 중 늦은 날 (설계 §1 — 구현 때 공고 원문 한 건과 대조).
 */
export function fromApplyhome(row, kind) {
  const address = text(row?.HSSPLY_ADRES);
  const title = text(row?.HOUSE_NM);
  const url = link(row?.PBLANC_URL) ?? link(row?.HMPG_ADRES);
  if (!address?.includes("익산") || !title || !url || !row?.HOUSE_MANAGE_NO) return null;
  const starts = [row.SPSPLY_RCEPT_BGNDE, row.RCEPT_BGNDE, row.SUBSCRPT_RCEPT_BGNDE, row.GNRL_RCEPT_BGNDE].map(ymd).filter(Boolean).sort();
  const ends = [row.SPSPLY_RCEPT_ENDDE, row.RCEPT_ENDDE, row.SUBSCRPT_RCEPT_ENDDE, row.GNRL_RCEPT_ENDDE].map(ymd).filter(Boolean).sort();
  return {
    id: `applyhome:${row.HOUSE_MANAGE_NO}-${row.PBLANC_NO ?? ""}`,
    source: "청약홈",
    kind,
    typeLabel: kind,
    title,
    supplier: text(row.BSNS_MBY_NM),
    announcedAt: ymd(row.RCRIT_PBLANC_DE),
    applyStart: starts[0] ?? null,
    applyEnd: ends.at(-1) ?? null,
    url,
    address,
  };
}

/** 같은 공고는 한 번, 정정공고가 대신한 공고 · 끝난 공고 · 기간이 없는 공고는 뺀다 */
export function mergeNotices(list, today) {
  const replaced = new Set(list.map((n) => n?.replaces).filter(Boolean));
  const seen = new Set();
  const out = [];
  for (const n of list) {
    if (!n || replaced.has(n.id) || seen.has(n.id)) continue;
    if (!n.applyStart && !n.applyEnd) continue;
    if (n.applyEnd && n.applyEnd < today) continue;
    seen.add(n.id);
    const { replaces: _replaces, ...notice } = n;
    out.push(notice);
  }
  return out;
}

export function formatSnapshot(notices, generatedAt) {

async function myhomeAll(fetchJson, key, op) {
  const rows = [];
  for (let page = 1; page <= MAX_PAGES; page++) {
    const u = new URL(`${MYHOME}/${op}`);
    u.searchParams.set("serviceKey", key);
    u.searchParams.set("brtcCode", MYHOME_BRTC);
    u.searchParams.set("signguCode", MYHOME_SIGNGU);
    u.searchParams.set("pageNo", String(page));
    u.searchParams.set("numOfRows", String(ROWS));
    const got = myhomeRows(await fetchJson(u.toString()));
    rows.push(...got);
    if (got.length < ROWS) break;
  }
  return rows;
}

async function applyhomeAll(fetchJson, key, op) {
  const rows = [];
  for (let page = 1; page <= MAX_PAGES; page++) {
    const u = new URL(`${APPLYHOME}/${op}`);
    u.searchParams.set("serviceKey", key);
    u.searchParams.set("page", String(page));
    u.searchParams.set("perPage", String(ROWS));
    u.searchParams.set("cond[SUBSCRPT_AREA_CODE_NM::EQ]", "전북");
    const json = await fetchJson(u.toString());
    const got = Array.isArray(json?.data) ? json.data : [];
    rows.push(...got);
    if (got.length < ROWS || rows.length >= Number(json?.totalCount ?? 0)) break;
  }
  return rows;
}
```

```ts
export const NOTICE_KINDS: NoticeKind[] = ["공공임대", "공공분양", "아파트 분양", "무순위 · 잔여"];

/** 공고를 알릴 사는 곳 — 알림함 · 휴대폰 알림 모두 이 지역에 사는(답한) 사람에게만 */
export const NOTICE_REGION = "전북특별자치도 익산시";

export type NoticeStatus = "open" | "upcoming" | "ended";

export function noticeStatus(n: HousingNotice, today: string): NoticeStatus {
  if (n.applyEnd && n.applyEnd < today) return "ended";
  if (n.applyStart && today < n.applyStart) return "upcoming";
  return "open";
}

const byEnd = (a: HousingNotice, b: HousingNotice) => (a.applyEnd ?? "9999-12-31").localeCompare(b.applyEnd ?? "9999-12-31");
const byStart = (a: HousingNotice, b: HousingNotice) => (a.applyStart ?? "").localeCompare(b.applyStart ?? "");

/** 접수 중(마감 가까운 순) · 접수 예정(시작 가까운 순). 끝난 공고는 뺀다 — 스냅숏이 하루 늦을 수 있다. */
export function groupNotices(notices: readonly HousingNotice[], today: string, kind: NoticeKind | null = null) {
  const list = kind ? notices.filter((x) => x.kind === kind) : [...notices];
  return {
    open: list.filter((x) => noticeStatus(x, today) === "open").sort(byEnd),
    upcoming: list.filter((x) => noticeStatus(x, today) === "upcoming").sort(byStart),
  };
}

export function kindsPresent(notices: readonly HousingNotice[], today: string): NoticeKind[] {
  const live = new Set(notices.filter((x) => noticeStatus(x, today) !== "ended").map((x) => x.kind));
  return NOTICE_KINDS.filter((k) => live.has(k));
}

const DAY = 86_400_000;

/** components/DeadlineBadge 가 아는 말(lib/preparation.ts deadlineLabel 과 같은 말) */
export function noticeDeadlineLabel(n: HousingNotice, today: string): string {
  if (noticeStatus(n, today) === "upcoming") return "신청 예정";
  if (n.applyEnd === null) return "상시";
  const days = Math.round((Date.parse(n.applyEnd) - Date.parse(today)) / DAY);
  return days === 0 ? "오늘 마감" : `D-${days}`;
}

export function homeNotices(notices: readonly HousingNotice[], today: string): HousingNotice[] {
  const { open, upcoming } = groupNotices(notices, today);
  return [...open, ...upcoming].slice(0, 3);
}

export function snapshotDateLabel(generatedAt: string | null): string | null {
  if (!generatedAt) return null;
  const [, m, d] = generatedAt.slice(0, 10).split("-");
  return `${Number(m)}월 ${Number(d)}일`;
}
```

```ts
export const NEW_NOTICE_DAYS = 14;

export interface NoticeInboxItem {
  kind: "notice";
  key: string;
  notice: HousingNotice;
  days: number;
}

/** 새 청약 공고 — 사는 곳이 익산(답한 것)일 때만. '새 지원금'과 같은 규칙(11/08 결정) */
export function noticeInboxItems(notices: readonly HousingNotice[], answers: DiscoveryAnswers, todayISO: string): NoticeInboxItem[] {
  if (!answers.region || !policyAppliesToRegion(NOTICE_REGION, answers.region)) return [];
  const out: NoticeInboxItem[] = [];
  for (const notice of notices) {
    if (!notice.announcedAt || noticeStatus(notice, todayISO) === "ended") continue;
    const days = daysBetween(notice.announcedAt, todayISO);
    if (days < 0 || days >= NEW_NOTICE_DAYS) continue;
    out.push({ kind: "notice", key: `notice:${notice.id}`, notice, days });
  }
  return out.sort((a, b) => a.days - b.days);
}
```

```ts
  const res = await fetchImpl(url, { signal: AbortSignal.timeout(10_000), headers: { "cache-control": "no-cache" } });
```

```ts
/** 새 청약 공고 (11/22). 공고 목록은 정책과 따로 읽는다 — 하나가 실패해도 다른 하나는 돈다. */
async function announceNewNotices(deps: NotifierDeps, directory: Directory, summary: RunSummary, now: Date): Promise<NoticeSummary> {
  const result: NoticeSummary = { outcome: "no-new", newNotices: 0, recipients: 0 };
  let notices: HousingNotice[];
  try {
    notices = await deps.fetchNotices();
  } catch {
    return { ...result, outcome: "fetch-failed" };
  }
  const announced = await deps.push.getAnnouncedNotices();
  if (announced === null) {
    // 첫 실행: 지금 있는 공고를 한꺼번에 알리지 않는다
    await deps.push.saveAnnouncedNotices(new Set(notices.map((n) => n.id)), now);
    return { ...result, outcome: "initialized" };
  }
  const { toSend, toMark } = selectNewNotices(notices, announced, kst(now).date);
  if (toMark.length === 0) return result;
  result.newNotices = toSend.length;
  if (toSend.length > 0) {
    const message = composeNoticeMessage(toSend);
    for (const [userId, devices] of await directory.byUser()) {
      try {
        const item = await deps.users.get(userId);
        if (!item || !item.data.consentVersion || !livesInNoticeRegion(item.data.answers.values)) continue;
        result.recipients += 1;
        await deliver(deps, directory, summary, userId, devices, message);
      } catch (err) {
        skipUser(summary, devices, err);
      }
    }
  }
  // 보낸 '뒤에' 적는다 — 새 정책 알림과 같은 이유
  await deps.push.saveAnnouncedNotices(new Set([...announced, ...toMark]), now);
  return { ...result, outcome: toSend.length > 0 ? "sent" : "no-new" };
}
```
