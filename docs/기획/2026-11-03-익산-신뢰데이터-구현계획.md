# 익산 신뢰 데이터 구현 계획

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 익산 사용자에게 보이는 모든 항목이 원문 대조일·담당자 확인일·누르면 걸리는 문의 전화를 갖추고, 지역 상담 창구를 보여준다.

**Architecture:** 세 데이터 타입에 `contact`·`nextRound` 필드를 직접 추가하고(null = 미확인), 지역 상담 창구는 `data/consult-centers.json` 에 두어 기존 `policyAppliesToRegion` 접두사 규칙으로 매칭한다. 화면은 공통 컴포넌트 `ContactInfo` 하나로 상세·신청 준비·결과 카드에 그린다. 익산의 신뢰 기준은 데이터 불변식 테스트(`iksan-trust.test.ts`)가 지킨다.

**Tech Stack:** Next.js 15 App Router (static export), TypeScript, Tailwind 4, Vitest + Testing Library (jsdom), lucide-react

**Spec:** `docs/기획/2026-11-03-익산-신뢰데이터-설계.md`

## Global Constraints

- F0-5 추정 금지: 확인 못 한 값은 `null`. 그럴듯한 번호·날짜를 채우지 않는다
- 원문 대조: 온통청년 설명만 보고 채우지 않는다. 공고 원문까지 연다
- `notes` 는 내부 메모다. 화면에 그리지 않는다 (PR #115)
- `app/result/BenefitResultCard.tsx`, `lib/benefit-result.ts`, `app/result/ResultState.tsx` 는 건드리지 않는다
- 공통 컴포넌트는 `app/components/index.ts` 로만 export·import 한다
- 날짜 표시는 `formatDotDate` (`2026.09.03`)
- 전화 형식: `^(0\d{1,2}-\d{3,4}-\d{4}|1\d{3}-\d{4})$`
- 익산 region 값: `"전북특별자치도 익산시"`
- 터치 타깃 최소 44px (`min-h-11`)
- push·PR 은 사용자가 요청할 때만

```tsx
import type { ContactInfo as Contact } from "@/lib/types";
import { ExternalLink, ICON_SM, Phone } from "./icons";

export const telHref = (phone: string) => `tel:${phone.replaceAll("-", "")}`;

/**
 * 담당 연락처. 정책 상세·신청 준비·결과 카드가 같은 모양으로 쓴다.
 *
 * null 이면 그리지 않는다 — "문의: 확인 필요" 같은 빈 칸은 신뢰를 깎는다.
 * 확인 못 한 항목은 화면에서 조용히 빠지고, 출처 링크가 그 자리를 대신한다.
 */
export function ContactInfo({ contact, variant }: { contact: Contact | null; variant: "block" | "inline" }) {
  if (!contact) return null;
  const phoneLink = contact.phone && (
    <a
      href={telHref(contact.phone)}
      aria-label={`문의 전화 ${contact.phone}`}
      className="focus-ring inline-flex min-h-11 items-center gap-1.5 rounded-control font-bold text-brand-700 underline hover:text-brand-800"
    >
      <Phone size={ICON_SM - 2} aria-hidden="true" />
      {contact.phone}
    </a>
  );

  if (variant === "inline") {
    return (
      <p className="flex flex-wrap items-center gap-x-1.5 text-sm text-ink-600">
        <span>문의 · {contact.department}</span>
        {phoneLink}
      </p>
    );
  }

  return (
    <section aria-label="문의" className="rounded-control bg-surface px-4 py-3.5 shadow-card">
      <h2 className="text-sm font-bold leading-[22px] text-ink-900">문의</h2>
      <p className="mt-1 text-xs text-ink-600">{contact.department}</p>
      <div className="mt-1 flex flex-wrap items-center gap-x-4 text-sm">
        {phoneLink}
        {contact.url && (
          <a
            href={contact.url}
            target="_blank"
            rel="noopener noreferrer"
            className="focus-ring inline-flex min-h-11 items-center gap-1 rounded-control text-ink-600 underline hover:text-ink-900"
          >
            문의 페이지
            <ExternalLink size={ICON_SM - 2} aria-hidden="true" />
          </a>
        )}
      </div>
    </section>
  );
}
```

/**
 * 지역 상담 창구. 가까운 곳이 먼저 오게 regionScope 가 긴 순서로 — 익산시 창구가
 * 전국 콜센터보다 위에 있어야 한다.
 */
export function consultCentersForRegion(centers: ConsultCenterMeta[], region: string): ConsultCenterMeta[] {
  const scopeRank = (c: ConsultCenterMeta) => (c.regionScope === "전국" ? 0 : strip(c.regionScope).length);
  return centers
    .filter((c) => policyAppliesToRegion(c.regionScope, region))
    .sort((a, b) => scopeRank(b) - scopeRank(a));
}

```ts
import { describe, expect, it } from "vitest";
import policiesJson from "@/data/policies.json";
import loanJson from "@/data/loan-products.json";
import housingJson from "@/data/housing-supply.json";
import { housingSupplyForRegion, loanProductsForRegion, policiesForRegion } from "@/lib/region";
import type { HousingSupplyMeta, LoanProductMeta, PolicyMeta } from "@/lib/types";

/**
 * 익산의 신뢰 기준. 익산 사용자에게 보이는 모든 항목은
 * 원문 대조일·담당 연락처·실제 신청 페이지를 갖는다.
 *
 * 이 테스트가 깨지면 새로 넣은 항목이 기준을 못 채운 것이다. 기준을 낮추지 말고 데이터를 채운다.
 */
const IKSAN = "전북특별자치도 익산시";
const items = [
  ...policiesForRegion(policiesJson as PolicyMeta[], IKSAN),
  ...loanProductsForRegion(loanJson as LoanProductMeta[], IKSAN),
  ...housingSupplyForRegion(housingJson as HousingSupplyMeta[], IKSAN),
];

describe("익산 신뢰 기준", () => {
  it.each(items.map((x) => [x.id, x] as const))("%s — 원문 대조일이 있다", (_, item) => {
    expect(item.verifiedAt).not.toBeNull();
  });

  it.each(items.map((x) => [x.id, x] as const))("%s — 담당 연락처가 있다", (_, item) => {
    expect(item.contact).not.toBeNull();
  });

  it.each(items.map((x) => [x.id, x] as const))("%s — 신청 링크가 홈페이지 첫 화면이 아니다", (_, item) => {
    const url = new URL(item.applyUrl);
    expect(url.pathname === "/" && url.search === "", `${item.applyUrl}`).toBe(false);
  });

  it("데이터 파일 사이에 id 가 겹치지 않는다", () => {
    const ids = [
      ...(policiesJson as PolicyMeta[]).map((x) => x.id),
      ...(loanJson as LoanProductMeta[]).map((x) => x.id),
      ...(housingJson as HousingSupplyMeta[]).map((x) => x.id),
    ];
    const dup = ids.filter((id, i) => ids.indexOf(id) !== i);
    expect(dup).toEqual([]);
  });
});
```
